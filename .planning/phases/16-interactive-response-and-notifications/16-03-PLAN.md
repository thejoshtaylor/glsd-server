**useNotifications.ts** — Create new hook (~40 lines):

```typescript
import { useState, useCallback } from 'react'

export function useNotifications() {
  const [permission, setPermission] = useState<NotificationPermission>(
    typeof Notification !== 'undefined' ? Notification.permission : 'default'
  )

  const requestPermission = useCallback(async () => {
    if (typeof Notification === 'undefined') return 'denied' as const
    const result = await Notification.requestPermission()
    setPermission(result)
    return result
  }, [])

  const notify = useCallback((title: string, body: string) => {
    if (typeof Notification === 'undefined') return
    if (Notification.permission !== 'granted') return
    if (!document.hidden) return  // Only notify when tab is unfocused
    new Notification(title, { body, icon: '/vite.svg' })
  }, [])

  return { permission, requestPermission, notify }
}
```

Key points:
- `typeof Notification !== 'undefined'` guard for SSR safety.
- `document.hidden` check prevents notifications when tab is focused (RESEARCH.md pitfall 4).
- `requestPermission` only called from user gesture (NotificationSettings button), never on load.

**NotificationSettings.tsx** — Create new component (~50 lines):

```typescript
import { useNotifications } from '@/hooks/useNotifications'
import { Button } from '@/components/ui/button'
import { Bell, BellOff } from '@/lib/icons'

export function NotificationSettings() {
  const { permission, requestPermission } = useNotifications()

  // If Notification API not supported, don't render
  if (typeof Notification === 'undefined') return null

  return (
    <div className="flex items-center justify-between p-3 rounded-lg border border-border bg-card/50">
      <div className="flex items-center gap-2 text-sm">
        {permission === 'granted' ? (
          <Bell size={16} className="text-primary" />
        ) : (
          <BellOff size={16} className="text-muted-foreground" />
        )}
        <span className="text-foreground">Browser Notifications</span>
        <span className="text-muted-foreground text-xs">
          {permission === 'granted' && '(enabled)'}
          {permission === 'denied' && '(blocked by browser)'}
          {permission === 'default' && '(not enabled)'}
        </span>
      </div>
      {permission === 'default' && (
        <Button
          variant="outline"
          size="sm"
          onClick={requestPermission}
        >
          Enable
        </Button>
      )}
      {permission === 'denied' && (
        <span className="text-xs text-muted-foreground">
          Reset in browser settings
        </span>
      )}
      {permission === 'granted' && (
        <span className="text-xs text-green-400">Active</span>
      )}
    </div>
  )
}
```

Key points:
- Shows current permission state: granted (green "Active"), denied (hint to reset in browser), default (Enable button).
- Enable button calls `requestPermission()` which is a user gesture (pitfall 4 prevention).
- No auto-request on mount.
- Uses project icon pattern (lucide via icons.ts).
